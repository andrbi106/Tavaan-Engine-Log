## استفاده

```php
use Tavaan\Engine\Log\LogManager;
use Tavaan\Engine\Log\Logger;
use Tavaan\Engine\Log\LogLevel;
use Tavaan\Engine\Log\Handlers\FileHandler;
use Tavaan\Engine\Log\Handlers\StreamHandler;
use Tavaan\Engine\Log\Formatters\JsonFormatter;
use Tavaan\Engine\Log\Formatters\LineFormatter;

// از طریق Manager
$log = new LogManager([
    'level' => 'debug',
    'handlers' => [
        ['type' => 'file', 'path' => 'storage/logs/app.log', 'format' => 'json'],
    ],
]);

// ۸ سطح RFC 5424
$log->emergency('system.down');
$log->alert('action.needed', ['service' => 'db']);
$log->critical('critical.error');
$log->error('order.failed', ['order_id' => 42]);
$log->warning('rate.limit');
$log->notice('job.queued');
$log->info('user.login', ['user_id' => 1]);
$log->debug('debug.info', ['dump' => $var]);

// کانال جداگانه
$audit = $log->channel('audit');
$audit->info('user.role.changed', ['user' => 5, 'role' => 'admin']);

// Context (یک‌بار برای یک لاگ)
$log->withContext(['order_id' => $id])->info('order.paid');

// Shared context (برای کل درخواست)
$log->shareContext(['request_id' => 'req-123']);
$log->info('step.1');
$log->info('step.2');
// هر دو شامل request_id خواهند بود

// Direct logger
$logger = new Logger('app');
$logger->addHandler(new FileHandler('app.log', new JsonFormatter()));
$logger->info('hello');
$logger->flush();
```
